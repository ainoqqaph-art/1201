```bash
#設定使用者名稱與EMAIL
git config --global user.email email
git config --global user.name name
#查看設定列表
git config --list
#git初始化
git init

git add

git commit -m'要加入的訊息'

#查看狀態
git status

#查看紀錄
git log
git log --oneline

git reflog

#回復
git reset 版本編碼 --hard
